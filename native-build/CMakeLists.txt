cmake_minimum_required(VERSION 3.22.1)
project(trutranslate CXX)
set(LLAMA_DIR /home/sandbox/trutranslate/llama.cpp)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)

target_include_directories(trutranslate PRIVATE ${LLAMA_DIR}/include ${LLAMA_DIR}/ggml/include)
target_link_libraries(trutranslate PRIVATE llama android log)
